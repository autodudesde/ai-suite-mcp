# cat=mcpwrites/writestarget; type=options[Draft workspace=workspace,Live=live]; label=Write mode:How MCP tools write data. Draft workspace (default) = always a draft; if the user has none, a per-user workspace is created automatically (requires ext:workspaces), and a write is refused rather than silently sent to live. Live = always direct to live. ChEddi inherits this by default but can override it with its own chatWriteMode (cheddi).
mcpWriteMode = workspace

# cat=mcpwrites/writestarget; type=boolean; label=Allow raw HTML write:Allow MCP write tools to store raw markup verbatim in code editor fields (e.g. tt_content.bodytext of CType "html"). TYPO3 renders those fields unfiltered, so an agent - or a prompt injection - can place arbitrary HTML/JavaScript on the site. When disabled, a write that carries markup into such a field is rejected: nothing is stored and the existing value survives. ChEddi inherits this by default and can only tighten it with its own chatAllowRawHtmlWrite (cheddi).
mcpAllowRawHtmlWrite = 0
